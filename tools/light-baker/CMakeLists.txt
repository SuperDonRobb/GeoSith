cmake_minimum_required(VERSION 3.20)
project(GeoSithLightBaker LANGUAGES CXX)
add_executable(GeoSithLightBaker WIN32 GeoSithLightBaker.cpp ../../src/Main/edGeometry.c ../../src/Main/edLevelIO.c)
set_source_files_properties(../../src/Main/edGeometry.c ../../src/Main/edLevelIO.c PROPERTIES LANGUAGE CXX)
target_compile_features(GeoSithLightBaker PRIVATE cxx_std_20)
target_include_directories(GeoSithLightBaker PRIVATE ../../src/Main ../../src)
target_compile_definitions(GeoSithLightBaker PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(GeoSithLightBaker PRIVATE user32 shell32 ole32)
set_property(TARGET GeoSithLightBaker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_options(GeoSithLightBaker PRIVATE /STACK:16777216)
